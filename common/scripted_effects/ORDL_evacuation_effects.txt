	if = {
		limit = {
			is_ai = yes
			has_completed_focus = SOV_move_industry_to_the_urals
			has_political_power > 34
			has_equipment = { train_equipment > 19 }
		}

# scope = the state being evacuated, ROOT = SOV; PP is charged by the wrapper
ORDL_evac_apply_state_effect = {
	if = {
		limit = {
			NOT = { OR = { has_state_flag = SOV_evac_done_flag has_state_flag = SOV_evac_lost_flag } }
			controller = { original_tag = SOV }
		}
		set_variable = { ROOT.SOV_evac_left_af = arms_factory_level }
		set_variable = { ROOT.SOV_evac_left_ic = industrial_complex_level }
		if = {
			limit = { NOT = { OR = { arms_factory > 0 industrial_complex > 0 } } }
			set_state_flag = SOV_evac_done_flag
		}
		else = {
			if = {
				limit = {
					ROOT = {
						check_variable = { SOV_evac_paid > 0 }
					}
				}

				add_to_variable = { global.ordl_evac_out_af = arms_factory_level }
				add_to_variable = { global.ordl_evac_out_ic = industrial_complex_level }
				if = { limit = { state = 11 }
					add_to_variable = { global.ordl_evac_out_11 = arms_factory_level }
					add_to_variable = { global.ordl_evac_out_11 = industrial_complex_level }
				}
				if = { limit = { state = 12 }
					add_to_variable = { global.ordl_evac_out_12 = arms_factory_level }

		limit = { has_country_flag = SOV_evac_region_S3_active }
		clr_country_flag = SOV_evac_region_S3_active
		if = {
			limit = { check_variable = { SOV_evac_regions_active > 0 } }
			subtract_from_variable = { SOV_evac_regions_active = 1 }
		}
		subtract_from_variable = { SOV_evac_transit_civ = SOV_evac_region_S3_civ }
		subtract_from_variable = { SOV_evac_transit_mil = SOV_evac_region_S3_mil }
		set_variable = { SOV_evac_region_S3_civ = 0 }
		set_variable = { SOV_evac_region_S3_mil = 0 }
	}
	clr_country_flag = SOV_evac_region_S3_running
}
